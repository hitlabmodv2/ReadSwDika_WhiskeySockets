import axios from "axios";
import fs from "fs";
import { spawn } from "child_process";
import {
  proto,
  generateWAMessageFromContent,
  generateWAMessageContent
} from "@whiskeysockets/baileys";

const handler = async (m, { conn, text, usedPrefix, command }) => {
  // Mengunci apikey murni ke variabel global therasav
  const apikey = global.therasav || "Apikeymu";

  const quotedMini = {
    key: {
      participant: "0@s.whatsapp.net",
      remoteJid: "0@s.whatsapp.net",
      fromMe: false,
      id: "Shikimori"
    },
    message: {
      conversation: m.text || `${usedPrefix + command}`
    }
  };

  if (command === "ytqualitych") {
    if (!text) return;
    let videoUrl = text.trim();

    const msg = generateWAMessageFromContent(
      m.chat,
      {
        viewOnceMessage: {
          message: {
            interactiveMessage: proto.Message.InteractiveMessage.fromObject({
              body: proto.Message.InteractiveMessage.Body.fromObject({
                text: `🎥 *PILIH KUALITAS VIDEO — SHIKIMORI*\n\n_Silakan ketuk tombol di bawah untuk memilih resolusi video yang kamu inginkan:_`
              }),
              footer: proto.Message.InteractiveMessage.Footer.fromObject({
                text: "💗 Silakan pilih resolusi di bawah"
              }),
              nativeFlowMessage: proto.Message.InteractiveMessage.NativeFlowMessage.fromObject({
                buttons: [
                  { name: "quick_reply", buttonParamsJson: JSON.stringify({ display_text: "🎞️ 140p", id: `${usedPrefix}ytdlch 140 | ${videoUrl}` }) },
                  { name: "quick_reply", buttonParamsJson: JSON.stringify({ display_text: "🎞️ 360p", id: `${usedPrefix}ytdlch 360 | ${videoUrl}` }) },
                  { name: "quick_reply", buttonParamsJson: JSON.stringify({ display_text: "🎞️ 480p", id: `${usedPrefix}ytdlch 480 | ${videoUrl}` }) },
                  { name: "quick_reply", buttonParamsJson: JSON.stringify({ display_text: "🎞️ 720p", id: `${usedPrefix}ytdlch 720 | ${videoUrl}` }) },
                  { name: "quick_reply", buttonParamsJson: JSON.stringify({ display_text: "🎞️ 1080p", id: `${usedPrefix}ytdlch 1080 | ${videoUrl}` }) }
                ]
              })
            })
          }
        }
      },
      { quoted: quotedMini }
    );

    return await conn.relayMessage(m.chat, msg.message, {});
  }

  if (command === "ytdlch") {
    if (!text) return;
    let [type, videoUrl] = text.split("|").map(v => v ? v.trim() : "");
    if (!type || !videoUrl) return;

    await conn.sendMessage(m.chat, { react: { text: "📥", key: m.key } });

    if (!fs.existsSync('./tmp')) fs.mkdirSync('./tmp', { recursive: true });
    
    const timestamp = Date.now();
    const tempFile = `./tmp/raw_${timestamp}.${type === "mp3" ? "mp3" : "mp4"}`;
    const fixedFile = `./tmp/fixed_${timestamp}.mp4`;

    try {
      if (type === "mp3") {
        const { data: dlApi } = await axios.get(
          `https://api.theresav.biz.id/download/ytmp3/v5?url=${encodeURIComponent(videoUrl)}&apikey=${apikey}`
        );
        if (!dlApi?.status || !dlApi.result?.url) throw new Error("Gagal memuat jalur unduhan audio dari API.");

        const audio = dlApi.result;
        
        const audioRes = await axios.get(audio.url, { responseType: 'arraybuffer' });
        fs.writeFileSync(tempFile, Buffer.from(audioRes.data));

        await conn.sendMessage(m.chat, {
            audio: { url: tempFile },
            mimetype: 'audio/mpeg',
            fileName: `${audio.title || "audio"}.mp3`
        }, { quoted: m });

      } else {
        const { data: dlApi } = await axios.get(
          `https://api.theresav.biz.id/download/ytmp4/v2?url=${encodeURIComponent(videoUrl)}&quality=${type}&apikey=${apikey}`
        );
        if (!dlApi?.status || !dlApi.result?.url) throw new Error(`Resolusi ${type}p tidak didukung atau link kedaluwarsa.`);

        const video = dlApi.result;

        const resStream = await axios({ method: 'get', url: video.url, responseType: 'stream' });
        const writer = fs.createWriteStream(tempFile);
        resStream.data.pipe(writer);
        await new Promise((resolve, reject) => { writer.on('finish', resolve); writer.on('error', reject); });

        // Eksekusi FFmpeg (biar video bisa di play) 
        const ff = spawn('ffmpeg', [
          '-y',
          '-i', tempFile,
          '-c:v', 'copy',            
          '-c:a', 'aac',             
          '-movflags', '+faststart', 
          fixedFile
        ]);

        await new Promise((resolve, reject) => {
          ff.on('close', (code) => {
            if (code === 0) resolve();
            else reject(new Error('FFmpeg gagal menstabilisasi container video.'));
          });
        });

        let caption = `✨ *YOUTUBE DOWNLOADER — SHIKIMORI* ✨\n\n🎬 *File:* ${video.filename || "Video YouTube"}\n🎯 *Kualitas:* ${type}p\n\n💗 _Video berhasil distabilkan dan siap diputar!_`;

        await conn.sendMessage(m.chat, {
            video: { url: fixedFile },
            caption: caption,
            mimetype: 'video/mp4',
            fileName: video.filename || "video.mp4"
        }, { quoted: m });
      }

      await conn.sendMessage(m.chat, { react: { text: "✅", key: m.key } });

    } catch (err) {
      console.error(err);
      await conn.sendMessage(m.chat, { react: { text: "❌", key: m.key } });
      m.reply(`❌ *Sistem Downloader Error:* ${err.message}`);
    } finally {
      // AUTO CLEANUP SAMPAH STORAGE CONTAINER
      if (fs.existsSync(tempFile)) fs.unlinkSync(tempFile);
      if (fs.existsSync(fixedFile)) fs.unlinkSync(fixedFile);
    }
    return;
  }

  if (!text) {
    return m.reply(`🎵 *Format Salah kak!*\n\nContoh:\n*${usedPrefix + command}* selamat tinggal`);
  }

  await conn.sendMessage(m.chat, { react: { text: "🔍", key: m.key } });

  try {
    const { data: searchApi } = await axios.get(
      `https://api.theresav.biz.id/search/youtube?q=${encodeURIComponent(text)}&apikey=${apikey}`
    );

    if (!searchApi?.status || !searchApi.result?.length) {
      throw new Error("Lagu atau video tidak dapat ditemukan, Mane.");
    }

    const searchResults = searchApi.result.slice(0, 5);
    const cards = [];
    let rank = 1;

    for (const item of searchResults) {
      try {
        const imgContent = await generateWAMessageContent(
          { image: { url: item.imageUrl } },
          { upload: conn.waUploadToServer }
        );

        cards.push({
          body: proto.Message.InteractiveMessage.Body.fromObject({
            text: `📺 *Channel:* ${item.channel}\n⏳ *Durasi:* ${item.duration}`
          }),
          footer: proto.Message.InteractiveMessage.Footer.fromObject({
            text: `Pilihan Ke-${rank++} — Shikimori Music Core`
          }),
          header: proto.Message.InteractiveMessage.Header.fromObject({
            title: item.title,
            hasMediaAttachment: true,
            imageMessage: imgContent.imageMessage
          }),
          nativeFlowMessage: proto.Message.InteractiveMessage.NativeFlowMessage.fromObject({
            buttons: [
              {
                name: "quick_reply",
                buttonParamsJson: JSON.stringify({
                  display_text: "🎵 Download Audio",
                  id: `${usedPrefix}ytdlch mp3 | ${item.link}`
                })
              },
              {
                name: "quick_reply",
                buttonParamsJson: JSON.stringify({
                  display_text: "🎥 Download Video",
                  id: `${usedPrefix}ytqualitych ${item.link}`
                })
              }
            ]
          })
        });
      } catch (cardErr) {
        rank++;
        continue;
      }
    }

    if (cards.length === 0) throw new Error("Gagal merajut komponen canvas kartu biner.");

    const msg = generateWAMessageFromContent(
      m.chat,
      {
        viewOnceMessage: {
          message: {
            interactiveMessage: proto.Message.InteractiveMessage.fromObject({
              body: proto.Message.InteractiveMessage.Body.fromObject({
                text: `✨ *YOUTUBE CAROUSEL SEARCH — SHIKIMORI* ✨\n\n📝 Hasil kata kunci: *${text}*\n\n💗 _Silakan geser kartu ke kanan untuk melihat variasi hasil lainnya, lalu tekan tombol aksi pilihanmu!_`
              }),
              footer: proto.Message.InteractiveMessage.Footer.fromObject({
                text: "💗 Shikimori YouTube Canvas Engine"
              }),
              header: proto.Message.InteractiveMessage.Header.fromObject({
                hasMediaAttachment: false
              }),
              carouselMessage: proto.Message.InteractiveMessage.CarouselMessage.fromObject({
                cards
              })
            })
          }
        }
      },
      { quoted: quotedMini }
    );

    await conn.relayMessage(m.chat, msg.message, { messageId: msg.key.id });
    await conn.sendMessage(m.chat, { react: { text: "⚡", key: m.key } });

  } catch (e) {
    console.error(e);
    await conn.sendMessage(m.chat, { react: { text: "❌", key: m.key } });
    m.reply(`⚠️ *Error System:* ${e.message}`);
  }
};

handler.help = ["ytsearch <query>", "yts <query>"];
handler.tags = ["downloader"];
handler.command = /^(ytsearch|yts|ytdlch|ytqualitych)$/i;

export default handler;
