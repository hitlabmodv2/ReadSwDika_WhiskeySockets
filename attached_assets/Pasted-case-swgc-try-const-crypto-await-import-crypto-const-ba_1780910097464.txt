case 'swgc': {
 try {
 const crypto = await import('crypto');
 const baileys = await import('@whiskeysockets/baileys');
 const { downloadContentFromMessage, generateWAMessageContent, generateWAMessageFromContent } = baileys;

 // Mengambil quoted message dengan fallback struktur yang aman
 const quoted = m.message?.extendedTextMessage?.contextInfo?.quotedMessage || m.quoted?.message || m.quoted;
 
 let mediaType = null;
 let mediaContent = null;

 // Deteksi tipe media secara eksplisit dan aman
 if (quoted) {
 if (quoted.imageMessage) { 
 mediaType = 'imageMessage'; 
 mediaContent = quoted.imageMessage; 
 } else if (quoted.videoMessage) { 
 mediaType = 'videoMessage'; 
 mediaContent = quoted.videoMessage; 
 } else if (quoted.audioMessage) { 
 mediaType = 'audioMessage'; 
 mediaContent = quoted.audioMessage; 
 } else if (quoted.viewOnceMessageV2?.message?.imageMessage) { 
 mediaType = 'imageMessage'; 
 mediaContent = quoted.viewOnceMessageV2.message.imageMessage; 
 } else if (quoted.viewOnceMessageV2?.message?.videoMessage) { 
 mediaType = 'videoMessage'; 
 mediaContent = quoted.viewOnceMessageV2.message.videoMessage; 
 }
 }

 let content = {};

 // Jika mendeteksi adanya reply media (Foto/Video/Audio)
 if (mediaType && mediaContent) {
 const downloadType = mediaType.replace('Message', '');
 const stream = await downloadContentFromMessage(mediaContent, downloadType);
 let buffer = Buffer.from([]);
 
 for await (const chunk of stream) {
 buffer = Buffer.concat([buffer, chunk]);
 }

 if (buffer.length === 0) return reply('Gagal download media.');

 if (mediaType === 'imageMessage') {
 content.image = buffer;
 content.caption = text || mediaContent.caption || '';
 } else if (mediaType === 'videoMessage') {
 content.video = buffer;
 content.caption = text || mediaContent.caption || '';
 } else if (mediaType === 'audioMessage') {
 content.audio = buffer;
 content.mimetype = mediaContent.mimetype || 'audio/mp4';
 content.ptt = true;
 }
 } else {
 // Jika tidak ada reply media, kirim teks biasa
 if (!text) return reply('Reply foto/video/vn atau isi teks.');
 content.text = text;
 }

 // Background status default hitam
 content.backgroundColor = '#000000';

 const inside = await generateWAMessageContent(content, { upload: sock.waUploadToServer });
 const messageSecret = crypto.randomBytes(32);

 const msgContent = generateWAMessageFromContent(
 remoteJid,
 {
 groupStatusMessageV2: {
 message: {
 ...inside,
 messageContextInfo: {
 messageSecret
 }
 }
 }
 },
 { userJid: sock.user.id }
 );

 await sock.relayMessage(remoteJid, msgContent.message, { messageId: msgContent.key.id });
 } catch (e) {
 await reply('Gagal kirim SW GC:\n' + e.message);
 }
 break;
 }
